#!/usr/bin/env bash
set -euo pipefail

# 3-seed 5000-step pilots for the most important new models.
# Adjust STEPS/WALKERS in the environment if needed:
#   STEPS=5000 WALKERS=48 bash scripts/run_extended_pilots.sh

STEPS=${STEPS:-5000}
WALKERS=${WALKERS:-48}

for MODEL in g1dew g1demk g1dem34; do
  for SEED in 101 202 303; do
    python src/run_extended_mcmc.py --model $MODEL --seed $SEED --steps $STEPS --walkers $WALKERS --no-optimize --progress
  done
  python src/diagnose_extended_chains.py --model $MODEL
done

python src/compare_extended_models.py
